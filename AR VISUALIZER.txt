                src="tu_modelo.glb"

                ar-modes="webxr scene-viewer quick-look" 
                ar-scale="fixed" 
                camera-controls 
                shadow-intensity="1"
                alt="Modelo 3D interactivo">

  <script>
    const viewer = document.querySelector('#visor-3d');
    
    // Elementos del DOM
    const anclajeOrigen = viewer.querySelector('[slot="hotspot-origen"]');
    const anclajeX = viewer.querySelector('[slot="hotspot-x"]');
    const anclajeY = viewer.querySelector('[slot="hotspot-y"]');
    const anclajeZ = viewer.querySelector('[slot="hotspot-z"]');
    
    const etiquetaX = document.querySelector('#etiqueta-x');
    const etiquetaY = document.querySelector('#etiqueta-y');
    const etiquetaZ = document.querySelector('#etiqueta-z');
    
    const lineaX = document.querySelector('#linea-x');
    const lineaY = document.querySelector('#linea-y');
    const lineaZ = document.querySelector('#linea-z');

    // Cuando el modelo termine de cargar
    viewer.addEventListener('load', () => {
      // Obtenemos el tamaño real del modelo en metros
      const medidas = viewer.getDimensions();
      const centro = viewer.getBoundingBoxCenter();

      // Convertimos a milímetros para mayor precisión 
      etiquetaX.textContent = `X: ${(medidas.x * 1000).toFixed(1)} mm`;
      etiquetaY.textContent = `Y: ${(medidas.y * 1000).toFixed(1)} mm`;
      etiquetaZ.textContent = `Z: ${(medidas.z * 1000).toFixed(1)} mm`;

      // Posicionamos los puntos de anclaje en las esquinas del modelo
      // Origen (esquina inferior trasera izquierda)
      const origX = centro.x - (medidas.x / 2);
      const origY = centro.y - (medidas.y / 2);
      const origZ = centro.z - (medidas.z / 2);

      anclajeOrigen.dataset.position = `${origX} ${origY} ${origZ}`;
      anclajeX.dataset.position = `${origX + medidas.x} ${origY} ${origZ}`;
      anclajeY.dataset.position = `${origX} ${origY + medidas.y} ${origZ}`;
      anclajeZ.dataset.position = `${origX} ${origY} ${origZ + medidas.z}`;
    });

    // Función para actualizar las líneas SVG mientras el modelo gira
    const actualizarLineas = () => {
      // Obtenemos las coordenadas 2D en pantalla de los puntos 3D
      const rectOrigen = anclajeOrigen.getBoundingClientRect();
      const rectX = anclajeX.getBoundingClientRect();
      const rectY = anclajeY.getBoundingClientRect();
      const rectZ = anclajeZ.getBoundingClientRect();
      
      const visorRect = viewer.getBoundingClientRect();

      // Función auxiliar para sacar el centro del anclaje relativo al visor
      const getCentro = (rect) => ({
        x: rect.left - visorRect.left + (rect.width / 2),
        y: rect.top - visorRect.top + (rect.height / 2)
      });

      const pOrigen = getCentro(rectOrigen);
      const pX = getCentro(rectX);
      const pY = getCentro(rectY);
      const pZ = getCentro(rectZ);

      // Dibujamos la línea X
      lineaX.setAttribute('x1', pOrigen.x);
      lineaX.setAttribute('y1', pOrigen.y);
      lineaX.setAttribute('x2', pX.x);
      lineaX.setAttribute('y2', pX.y);
      etiquetaX.style.left = `${(pOrigen.x + pX.x) / 2}px`;
      etiquetaX.style.top = `${(pOrigen.y + pX.y) / 2}px`;

      // Dibujamos la línea Y
      lineaY.setAttribute('x1', pOrigen.x);
      lineaY.setAttribute('y1', pOrigen.y);
      lineaY.setAttribute('x2', pY.x);
      lineaY.setAttribute('y2', pY.y);
      etiquetaY.style.left = `${(pOrigen.x + pY.x) / 2}px`;
      etiquetaY.style.top = `${(pOrigen.y + pY.y) / 2}px`;

      // Dibujamos la línea Z
      lineaZ.setAttribute('x1', pOrigen.x);
      lineaZ.setAttribute('y1', pOrigen.y);
      lineaZ.setAttribute('x2', pZ.x);
      lineaZ.setAttribute('y2', pZ.y);
      etiquetaZ.style.left = `${(pOrigen.x + pZ.x) / 2}px`;
      etiquetaZ.style.top = `${(pOrigen.y + pZ.y) / 2}px`;
    };

    // Actualizamos las líneas cada vez que se mueva la cámara
    viewer.addEventListener('camera-change', actualizarLineas);
    // Y forzamos una actualización inicial cuando los anclajes estén listos
    window.addEventListener('resize', actualizarLineas);
    setTimeout(actualizarLineas, 500); // Pequeño retraso para asegurar el renderizado
  </script>
